cmake_minimum_required(VERSION 3.22.1)
project(ember_reader C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(SHERPA "${PROJECT_ROOT}/.deps/sherpa")
foreach(LIB sherpa-onnx-c-api onnxruntime)
    add_library(${LIB} SHARED IMPORTED)
    set_target_properties(${LIB} PROPERTIES IMPORTED_LOCATION "${SHERPA}/lib${LIB}.so")
endforeach()
add_library(ember_fast SHARED fast_jni.c question_prosody.cpp)
target_include_directories(ember_fast PRIVATE "${SHERPA}"
    "${PROJECT_ROOT}/.deps/signalsmith-stretch" "${PROJECT_ROOT}/.deps/signalsmith-linear/include")
target_compile_options(ember_fast PRIVATE -O3 -fvisibility=hidden)
target_link_options(ember_fast PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(ember_fast PRIVATE sherpa-onnx-c-api onnxruntime m log)
